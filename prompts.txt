<script>
  const STORAGE_KEY = "studyTrackerTasks_v1";

  const initialTasks = [
    { id: 1, title: "Чтение", hours: 2, done: true },
    { id: 2, title: "Практика", hours: 4, done: false },
    { id: 3, title: "Повторение", hours: 2, done: false }
  ];

  let tasks = loadTasks();
  let currentFilter = "all";

  const taskForm = document.getElementById("taskForm");
  const taskInput = document.getElementById("taskInput");
  const hoursInput = document.getElementById("hoursInput");
  const taskList = document.getElementById("taskList");

  function loadTasks() {
    try {
      const saved = localStorage.getItem(STORAGE_KEY);
      if (saved === null) return [...initialTasks];

      const parsed = JSON.parse(saved);

      if (!Array.isArray(parsed)) return [...initialTasks];

      return parsed.filter(task =>
        task &&
        typeof task.title === "string" &&
        typeof task.hours === "number" &&
        typeof task.done === "boolean"
      );
    } catch {
      return [...initialTasks];
    }
  }

  function saveTasks() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
    } catch (error) {
      console.warn("Не удалось сохранить задачи", error);
    }
  }

  function formatHours(value) {
    return Number(value.toFixed(2)).toLocaleString("ru-RU");
  }

  function render() {
    const totalHours = tasks.reduce((sum, task) => sum + task.hours, 0);
    const doneHours = tasks
      .filter(task => task.done)
      .reduce((sum, task) => sum + task.hours, 0);

    const percent = totalHours > 0
      ? Math.round(doneHours / totalHours * 100)
      : 0;

    document.getElementById("totalHours").textContent =
      formatHours(totalHours) + " ч";

    document.getElementById("doneHours").textContent =
      formatHours(doneHours) + " ч";

    document.getElementById("taskCount").textContent = tasks.length;

    document.getElementById("progressText").textContent = percent + "%";
    document.getElementById("progressBar").style.width = percent + "%";

    taskList.replaceChildren();

    const filteredTasks = tasks.filter(task => {
      if (currentFilter === "active") return !task.done;
      if (currentFilter === "done") return task.done;
      return true;
    });

    if (filteredTasks.length === 0) {
      const empty = document.createElement("div");
      empty.className = "empty";
      empty.textContent = tasks.length === 0
        ? "Пока нет задач. Добавь свою первую!"
        : "В этом фильтре задач нет.";
      taskList.appendChild(empty);
      return;
    }

    filteredTasks.forEach(task => {
      const row = document.createElement("div");
      row.className = "task" + (task.done ? " done" : "");

      const checkbox = document.createElement("input");
      checkbox.type = "checkbox";
      checkbox.checked = task.done;
      checkbox.setAttribute("aria-label", "Отметить задачу: " + task.title);

      checkbox.addEventListener("change", () => {
        task.done = checkbox.checked;
        saveTasks();
        render();
      });

      const main = document.createElement("div");
      main.className = "task-main";

      const title = document.createElement("div");
      title.className = "task-title";
      title.textContent = task.title;

      const hours = document.createElement("div");
      hours.className = "task-hours";
      hours.textContent = formatHours(task.hours) + " ч";

      main.append(title, hours);

      const removeBtn = document.createElement("button");
      removeBtn.className = "delete-btn";
      removeBtn.textContent = "×";
      removeBtn.title = "Удалить задачу";
      removeBtn.setAttribute("aria-label", "Удалить: " + task.title);

      removeBtn.addEventListener("click", () => {
        tasks = tasks.filter(item => item.id !== task.id);
        saveTasks();
        render();
      });

      row.append(checkbox, main, removeBtn);
      taskList.appendChild(row);
    });
  }

  taskForm.addEventListener("submit", event => {
    event.preventDefault();

    const title = taskInput.value.trim();
    const hours = Number(hoursInput.value);

    if (!title || !Number.isFinite(hours) || hours <= 0) {
      return;
    }

    tasks.push({
      id: Date.now() + Math.random(),
      title,
      hours,
      done: false
    });

    taskInput.value = "";
    hoursInput.value = "1";
    taskInput.focus();

    saveTasks();
    render();
  });

  document.querySelectorAll(".filter-btn").forEach(button => {
    button.addEventListener("click", () => {
      currentFilter = button.dataset.filter;

      document.querySelectorAll(".filter-btn").forEach(btn => {
        btn.classList.toggle("active", btn === button);
      });

      render();
    });
  });

  render();
</script>

# Proposal: Мини-трекер учебных задач на неделю

## Цель

Создать автономный мини-трекер учебных задач на неделю, позволяющий
добавлять, удалять, отмечать выполненные задачи, фильтровать список
и отслеживать суммарные часы и процент выполнения.

Трекер должен работать без CDN, внешних библиотек и API-ключей.

## Границы

В рамках задачи реализуется:

- Добавление учебных задач.
- Удаление задач.
- Переключение состояния выполнения.
- Валидация названия и количества часов.
- Фильтрация задач по состоянию.
- Подсчёт общего количества часов и прогресса.
- Адаптивный интерфейс для ширины 390 px.
- Автономная работа приложения без внешних зависимостей.

## Варианты реализации

### Вариант A

Один автономный файл `tracker-v2.html`, содержащий HTML, CSS и JavaScript.

### Вариант B

Разделение приложения на файлы:

- `index.html` — структура интерфейса.
- `styles.css` — стили.
- `app.mjs` — интерфейсная логика.
- `domain.mjs` — модель данных, правила и расчёты.
- `tests/acceptance.test.mjs` — приёмочные тесты.

## Не входит в задачу

- Серверная часть.
- Синхронизация между устройствами.
- Авторизация.
- Внешние API.
- Внешние библиотеки.
- CDN.
- API-ключи.
- Дополнительные функции, не описанные в требованиях.

## Необязательное улучшение

Сохранение задач в localStorage с обработкой недоступности
хранилища и продолжением работы в памяти.

# Specification: Учебный трекер

## 1. Управление задачами

### Requirement: Добавление задачи

Система MUST позволять добавлять учебные задачи с названием
и количеством часов.

- Название после trim MUST быть непустым.
- Часы MUST быть числом от 0.5 до 24 включительно.
- Часы MUST изменяться с шагом 0.5.
- Пустое значение часов MUST NOT трактоваться как 0.
- Новая задача MUST иметь уникальный id.
- Новая задача MUST иметь done=false.
- Некорректная задача MUST NOT добавляться.
- При ошибке MUST отображаться сообщение рядом с полем.
- При ошибке введённые значения формы MUST сохраняться.

#### Scenario: Добавление корректной задачи

WHEN пользователь вводит непустое название и допустимое количество
часов и подтверждает добавление

THEN задача добавляется в список
AND её состояние done равно false
AND форма очищается после успешного добавления.

#### Scenario: Пустое название

WHEN пользователь вводит пустое название или название,
состоящее только из пробелов

THEN задача не добавляется
AND отображается ошибка рядом с полем названия
AND введённые значения формы сохраняются.

#### Scenario: Пустые часы

WHEN пользователь не вводит значение часов

THEN задача не добавляется
AND отображается ошибка рядом с полем часов
AND введённые значения формы сохраняются.

#### Scenario: Недопустимые часы

WHEN пользователь вводит отрицательное число, число меньше 0.5,
число больше 24 или число, не соответствующее шагу 0.5

THEN задача не добавляется
AND отображается ошибка рядом с полем часов
AND введённые значения формы сохраняются.

## 2. Удаление задач

### Requirement: Удаление задачи

Система MUST позволять удалять существующие задачи.

#### Scenario: Удаление задачи

WHEN пользователь удаляет задачу

THEN задача удаляется из данных
AND список обновляется
AND суммарные часы и прогресс пересчитываются.

## 3. Выполнение задач

### Requirement: Переключение done

Система MUST позволять переключать состояние done
между true и false.

#### Scenario: Отметка задачи выполненной

WHEN пользователь отмечает задачу выполненной

THEN done становится true
AND прогресс пересчитывается.

#### Scenario: Снятие отметки выполнения

WHEN пользователь снимает отметку выполнения

THEN done становится false
AND прогресс пересчитывается.

## 4. Фильтрация

### Requirement: Фильтр списка

Система MUST поддерживать фильтры:

- Все.
- Активные.
- Готовые.

Фильтр MUST изменять только отображаемый список.

Фильтр MUST NOT изменять исходные данные задач,
суммарные часы или прогресс.

#### Scenario: Фильтр активных задач

WHEN включён фильтр «активные»

THEN отображаются только задачи с done=false

прогресс MUST быть равен 0%.

# Design: Учебный трекер

## 1. Архитектура

Приложение представляет собой автономный мини-трекер
с локальным управлением состоянием.

### Вариант A

Файл tracker-v2.html включает:

- HTML-разметку.
- CSS-стили.
- JavaScript.
- Модель данных.
- Валидацию.
- Расчёты.
- Обработчики событий.
- Отображение списка и итогов.

### Вариант B

- index.html — разметка.
- styles.css — стили и адаптивность.
- app.mjs — управление интерфейсом и событиями.
- domain.mjs — логика задач, валидация и расчёты.
- tests/acceptance.test.mjs — проверка приёмочных сценариев.

## 2. Модель данных

Задача:

- id — уникальный идентификатор.
- title — название задачи.
- hours — количество часов.
- done — статус выполнения (boolean).

Пример начального состояния:

[
  { id: 1, title: "Чтение", hours: 2, done: true },
  { id: 2, title: "Практика", hours: 4, done: false },
  { id: 3, title: "Повторение", hours: 2, done: false }
]

ID должны быть уникальными. Значения ID в примере
представлены условно.

## 3. Управление состоянием

Состояние приложения содержит:

- Массив задач.
- Текущий фильтр: all, active или done.
- Значения полей формы.
- Ошибки валидации.
- Состояние доступности localStorage (если реализовано).

Операции добавления, удаления и переключения done
изменяют данные задач.

Фильтр изменяет только представление списка.

## 4. Валидация

Название:

- Удаление пробелов по краям с помощью trim.
- Проверка на непустое значение.

Часы:

- Значение обязательно.
- Значение должно быть числом.
- Минимум: 0.5.
- Максимум: 24.
- Шаг: 0.5.

Ошибки отображаются рядом с соответствующим полем.
При ошибке форма не сбрасывается.

## 5. Расчёты

### Общие часы

totalHours = сумма hours всех задач.

### Выполненные часы

doneHours = сумма hours задач, у которых done=true.

### Прогресс

Если totalHours равно 0:

progress = 0.

Иначе:

progress = round(100 * doneHours / totalHours).

Расчёты выполняются по всему массиву задач,
а не по отфильтрованному списку.

## 6. Фильтрация

- all — отображаются все задачи.
- active — отображаются задачи с done=false.
- done — отображаются задачи с done=true.

Фильтрация не изменяет массив задач и не влияет на расчёты.

## 7. Начальные расчёты

Общее количество часов:

2 + 4 + 2 = 8.

Выполненные часы:

2.

Прогресс:

round(100 * 2 / 8) = 25%.

После выполнения «Практики»:

Выполненные часы:

2 + 4 = 6.

Прогресс:

round(100 * 6 / 8) = 75%.

## 8. Интерфейс

Интерфейс содержит:

- Заголовок трекера.
- Поле названия задачи.
- Поле часов.
- Кнопку добавления.
- Список задач.
- Управление выполнением.
- Кнопки удаления.
- Фильтры «Все», «Активные», «Готовые».
- Суммарные часы.
- Выполненные часы.
- Процент выполнения.

Интерфейс должен быть адаптивным для ширины 390 px.
Горизонтальная прокрутка недопустима.

## 9. Необязательное сохранение

При реализации localStorage:

- Сохранять актуальные задачи.
- Восстанавливать сохранённые задачи при запуске.
- Обрабатывать ошибки чтения и записи.
- При недоступности хранилища продолжать работу в памяти.
- Показывать понятное сообщение о недоступности хранилища.

## 10. Ограничения

- Не использовать CDN.
- Не использовать внешние библиотеки.
- Не использовать API-ключи.
- Не реализовывать серверную часть.
- Не расширять функциональность сверх требований.
